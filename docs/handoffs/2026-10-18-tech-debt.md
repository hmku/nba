Golden script (run before/after each refactor; diff the `*.norm` files and `proj.sha`):
```bash
#!/usr/bin/env bash
# usage: golden.sh <worktree> <outdir>
set -euo pipefail
WT=$1; OUT=$2; rm -rf "$OUT"; mkdir -p "$OUT"
export PYTHONPATH=$WT/src
DB=$OUT/golden.db; rm -f $DB
export FANTASY_DB_URL=sqlite:///$DB
unset FANTASY_DRAFT_ID FANTASY_RUN_ID || true
F="python -m fantasy.cli"
cd $WT
$F settings show --json > $OUT/settings.json
$F values build --no-save --top 60 --json > $OUT/values.json
$F values build --json > /dev/null   # saved run needed by draft cmds
$F draft init --draft-id mock-g --mock --json > $OUT/init.json
export FANTASY_DRAFT_ID=mock-g
$F draft recommend --no-save --json > $OUT/rec1.json
$F draft queue --top 60 --json > $OUT/queue.json
ADP=$(python - <<'PY'
import pandas as pd
df=pd.read_csv("data/processed/projections_2026_27.csv")
df=df.assign(a=df.adp.fillna(9999)).sort_values(["a","player_id"])
print(" ".join(df.player_id.head(80)))
PY
)
fill() { # from to
  local n=$1; local to=$2
  for p in $ADP; do
    [ $n -gt $to ] && break
    if $F draft pick "$p" --overall $n --json > /dev/null 2>&1; then n=$((n+1)); fi
  done
}
top() { python -c "import json,sys;d=json.load(open('$1'));r=d.get('recommendations') or d.get('candidates');print(r[0]['player_id'])"; }
$F draft pick "$(top $OUT/rec1.json)" --overall 1 --json > /dev/null
fill 2 23
$F draft recommend --no-save --json > $OUT/rec24.json
$F draft pick "$(top $OUT/rec24.json)" --overall 24 --json > /dev/null
fill 25 35
$F draft recommend --no-save --json > $OUT/rec36.json
$F draft board --json > $OUT/board.json
unset FANTASY_DRAFT_ID
$F draft simulate --sims 3 --seed 0 --no-save --json > $OUT/simulate.json
$F sim season --draft-id mock-g --complete-rosters --sims 200 --seed 0 --no-save --json > $OUT/simseason.json 2>$OUT/simseason.err || echo "simseason rc=$?" >> $OUT/simseason.err
$F data build-projections --json > $OUT/buildproj.json
sha256sum data/processed/projections_2026_27.csv | cut -c1-64 > $OUT/proj.sha
git -C $WT checkout -q -- data/processed 2>/dev/null || true
python "$(dirname "$0")/norm.py" $OUT
```
Normaliser `norm.py`: drops keys matching `run_id|model_run_id|created_at|updated_at|generated_at|as_of|git_sha|git|elapsed.*|.*_seconds|timestamp|rec_id|id|started_at|fetched_at|built_at|ts|duration.*` recursively and writes sorted JSON.
